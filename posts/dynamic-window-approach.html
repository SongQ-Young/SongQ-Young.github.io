动态窗口法(DWA)局部路径规划实战

📅 2024-11-15 ✍️ SongQ-Young 🏷️ 局部规划
局部规划 DWA 避障

1. 引言

动态窗口法(Dynamic Window Approach, DWA)是一种广泛应用于移动机器人局部路径规划的实时避障算法。该算法由Dieter Fox等人于1997年提出，其核心思想是在速度空间中采样可行的速度组合，通过评估每条轨迹的优劣来选择最佳运动指令。DWA算法具有计算效率高、能够考虑机器人动力学约束等优点，在各类移动机器人平台上得到了广泛应用。

本文将深入探讨DWA算法的原理与实现，包括速度空间采样、轨迹评估函数设计、参数调优策略以及狭窄空间处理等关键技术点，并提供详细的C++代码实现。

2. DWA算法基本原理

2.1 算法核心思想

DWA算法的核心是在机器人当前速度附近的动态窗口内搜索可行的速度对(v, ω)，其中v为线速度，ω为角速度。算法通过以下步骤选择最优速度：

  • 速度采样: 在动态窗口内采样多组速度对
  • 轨迹预测: 根据每组速度预测未来一段时间的运动轨迹
  • 轨迹评估: 使用多目标评估函数对每条轨迹打分
  • 速度选择: 选择得分最高的速度作为控制输出

2.2 动态窗口定义

动态窗口是由以下约束条件共同确定的速度空间区域： 

struct VelocityConstraints {
    // 1. 机器人物理限制
    double max_v;      // 最大线速度
    double min_v;      // 最小线速度
    double max_w;      // 最大角速度
    double min_w;      // 最小角速度

    // 2. 加速度限制（动态窗口的核心）
    double max_accel_v;   // 最大线加速度
    double max_accel_w;   // 最大角加速度

    // 3. 当前速度
    double current_v;
    double current_w;

    // 计算动态窗口
    void calculate_dynamic_window(double dt,
                                  double& v_min, double& v_max,
                                  double& w_min, double& w_max) {
        // 考虑加速度约束
        v_min = std::max(min_v, current_v - max_accel_v * dt);
        v_max = std::min(max_v, current_v + max_accel_v * dt);
        w_min = std::max(min_w, current_w - max_accel_w * dt);
        w_max = std::min(max_w, current_w + max_accel_w * dt);
    }
};

2.3 算法流程

// DWA算法主流程
struct DWAConfig {
    double predict_time;        // 轨迹预测时间 (s)
    double dt;                  // 时间步长 (s)
    double v_resolution;        // 线速度采样分辨率
    double w_resolution;        // 角速度采样分辨率

    // 评估函数权重
    double heading_weight;      // 朝向目标权重
    double distance_weight;     // 障碍物距离权重
    double velocity_weight;     // 速度权重
};

class DWAPlanner {
public:
    std::pair compute_velocity(
        const RobotState& state,
        const std::vector& obstacles,
        const Point& goal) {

        // 1. 计算动态窗口
        auto [v_min, v_max, w_min, w_max] =
            calculate_dynamic_window(state);

        // 2. 速度空间采样
        std::vector velocity_samples;
        for (double v = v_min; v <= v_max; v += config_.v_resolution) {
            for (double w = w_min; w <= w_max; w += config_.w_resolution) {
                velocity_samples.push_back({v, w});
            }
        }

        // 3. 评估每个速度对应的轨迹
        double best_score = -INFINITY;
        Velocity best_velocity;

        for (const auto& vel : velocity_samples) {
            // 预测轨迹
            Trajectory traj = predict_trajectory(state, vel);

            // 检查碰撞
            if (is_collision(traj, obstacles)) {
                continue;
            }

            // 计算得分
            double score = evaluate_trajectory(traj, goal, vel);

            if (score > best_score) {
                best_score = score;
                best_velocity = vel;
            }
        }

        return {best_velocity.v, best_velocity.w};
    }
};

3. 速度空间与轨迹预测

3.1 差分驱动模型

对于差分驱动机器人，运动学模型如下： 

struct RobotState {
    double x, y;      // 位置
    double theta;     // 朝向角
    double v, w;      // 线速度和角速度
};

// 差分驱动运动学模型
RobotState motion_model(const RobotState& state,
                        double v, double w, double dt) {
    RobotState next_state = state;

    if (std::abs(w) < 1e-6) {
        // 直线运动
        next_state.x += v * std::cos(state.theta) * dt;
        next_state.y += v * std::sin(state.theta) * dt;
    } else {
        // 圆弧运动
        double radius = v / w;
        next_state.x += radius * (std::sin(state.theta + w * dt) -
                                  std::sin(state.theta));
        next_state.y += radius * (-std::cos(state.theta + w * dt) +
                                  std::cos(state.theta));
        next_state.theta += w * dt;
    }

    next_state.v = v;
    next_state.w = w;

    return next_state;
}

3.2 轨迹生成

struct TrajectoryPoint {
    double x, y, theta;
    double v, w;
    double time;
};

using Trajectory = std::vector;

Trajectory predict_trajectory(const RobotState& state,
                               double v, double w,
                               double predict_time,
                               double dt) {
    Trajectory trajectory;
    RobotState current = state;

    for (double t = 0; t <= predict_time; t += dt) {
        TrajectoryPoint point;
        point.x = current.x;
        point.y = current.y;
        point.theta = current.theta;
        point.v = v;
        point.w = w;
        point.time = t;

        trajectory.push_back(point);

        // 预测下一状态
        current = motion_model(current, v, w, dt);
    }

    return trajectory;
}

3.3 碰撞检测

class CollisionChecker {
private:
    double robot_radius_;
    double safety_margin_;

public:
    CollisionChecker(double radius, double margin = 0.1)
        : robot_radius_(radius), safety_margin_(margin) {}

    bool is_collision(const Trajectory& trajectory,
                     const std::vector& obstacles) {
        double safe_distance = robot_radius_ + safety_margin_;

        for (const auto& point : trajectory) {
            for (const auto& obstacle : obstacles) {
                double dist = obstacle.distance_to_point(point.x, point.y);
                if (dist < safe_distance) {
                    return true;  // 发生碰撞
                }
            }
        }

        return false;
    }

    // 计算轨迹到最近障碍物的距离
    double min_obstacle_distance(const Trajectory& trajectory,
                                 const std::vector& obstacles) {
        double min_dist = INFINITY;

        for (const auto& point : trajectory) {
            for (const auto& obstacle : obstacles) {
                double dist = obstacle.distance_to_point(point.x, point.y);
                min_dist = std::min(min_dist, dist);
            }
        }

        return min_dist;
    }
};

4. 轨迹评估与评分函数

4.1 多目标评估函数

DWA算法使用多目标评估函数对轨迹进行评分，典型的评估函数包括三个部分： 

class TrajectoryEvaluator {
private:
    DWAConfig config_;

public:
    double evaluate(const Trajectory& trajectory,
                   const Point& goal,
                   const std::vector& obstacles,
                   double v, double w) {
        // 1. 目标朝向得分（heading）
        double heading_score = calculate_heading_score(trajectory, goal);

        // 2. 障碍物距离得分（clearance）
        double distance_score = calculate_distance_score(trajectory, obstacles);

        // 3. 速度得分（velocity）
        double velocity_score = calculate_velocity_score(v);

        // 加权求和
        double total_score =
            config_.heading_weight * heading_score +
            config_.distance_weight * distance_score +
            config_.velocity_weight * velocity_score;

        return total_score;
    }

private:
    // 计算朝向得分：轨迹末端朝向与目标方向的一致性
    double calculate_heading_score(const Trajectory& trajectory,
                                   const Point& goal) {
        if (trajectory.empty()) return 0.0;

        const auto& end_point = trajectory.back();

        // 计算目标方向角
        double goal_angle = std::atan2(goal.y - end_point.y,
                                       goal.x - end_point.x);

        // 计算角度差
        double angle_diff = normalize_angle(goal_angle - end_point.theta);

        // 转换为得分：角度差越小得分越高
        return M_PI - std::abs(angle_diff);
    }

    // 计算障碍物距离得分：与障碍物的最小距离
    double calculate_distance_score(const Trajectory& trajectory,
                                   const std::vector& obstacles) {
        CollisionChecker checker(robot_radius_);
        double min_dist = checker.min_obstacle_distance(trajectory, obstacles);

        // 距离越大得分越高，但有上限
        const double max_score_distance = 2.0;  // 超过2米得分不再增加
        return std::min(min_dist, max_score_distance);
    }

    // 计算速度得分：鼓励较高的前进速度
    double calculate_velocity_score(double v) {
        // 线速度越大得分越高
        return std::abs(v);
    }

    // 角度归一化到[-π, π]
    double normalize_angle(double angle) {
        while (angle > M_PI) angle -= 2.0 * M_PI;
        while (angle < -M_PI) angle += 2.0 * M_PI;
        return angle;
    }
};

4.2 改进的评估函数

在实际应用中，可以添加更多评估维度以提高性能： 

class AdvancedEvaluator : public TrajectoryEvaluator {
public:
    double evaluate_advanced(const Trajectory& trajectory,
                            const Point& goal,
                            const std::vector& obstacles,
                            double v, double w,
                            const Trajectory& global_path) {
        double basic_score = evaluate(trajectory, goal, obstacles, v, w);

        // 4. 路径跟踪得分：与全局路径的偏差
        double path_following_score = calculate_path_following_score(
            trajectory, global_path);

        // 5. 平滑度得分：惩罚剧烈的速度变化
        double smoothness_score = calculate_smoothness_score(v, w);

        // 6. 目标距离得分：接近目标时的额外奖励
        double goal_distance_score = calculate_goal_distance_score(
            trajectory, goal);

        return basic_score +
               0.5 * path_following_score +
               0.3 * smoothness_score +
               0.8 * goal_distance_score;
    }

private:
    double calculate_path_following_score(
        const Trajectory& trajectory,
        const Trajectory& global_path) {
        // 计算轨迹与全局路径的平均距离
        double total_deviation = 0.0;

        for (const auto& point : trajectory) {
            double min_dist = INFINITY;
            for (const auto& path_point : global_path) {
                double dist = std::hypot(point.x - path_point.x,
                                        point.y - path_point.y);
                min_dist = std::min(min_dist, dist);
            }
            total_deviation += min_dist;
        }

        double avg_deviation = total_deviation / trajectory.size();
        // 偏差越小得分越高
        return 1.0 / (1.0 + avg_deviation);
    }

    double calculate_smoothness_score(double v, double w) {
        // 惩罚与当前速度差异过大的速度指令
        double v_diff = std::abs(v - current_velocity_.v);
        double w_diff = std::abs(w - current_velocity_.w);

        return 1.0 / (1.0 + v_diff + w_diff);
    }

    double calculate_goal_distance_score(const Trajectory& trajectory,
                                        const Point& goal) {
        if (trajectory.empty()) return 0.0;

        const auto& end_point = trajectory.back();
        double dist = std::hypot(goal.x - end_point.x,
                                goal.y - end_point.y);

        // 距离目标越近得分越高
        return 1.0 / (1.0 + dist);
    }
};

5. 参数调优策略

5.1 关键参数说明

DWA算法的性能很大程度上取决于参数配置，以下是关键参数及其影响：

参数 典型值 影响 调优建议
predict_time 1.0~3.0s 预测时间越长，规划越保守 高速场景增大，狭窄空间减小
v_resolution 0.01~0.1 m/s 采样密度，影响计算量 根据实时性要求调整
w_resolution 0.05~0.2 rad/s 角速度采样密度 转弯场景需要更密集采样
heading_weight 0.5~2.0 朝向目标的重要性 开阔环境增大
distance_weight 1.0~3.0 避障的重要性 密集障碍物环境增大
velocity_weight 0.1~1.0 速度偏好 效率优先时增大

5.2 自适应参数调整

class AdaptiveDWA {
private:
    DWAConfig base_config_;

    // 根据环境动态调整参数
    DWAConfig adapt_parameters(const RobotState& state,
                               const std::vector& obstacles,
                               const Point& goal) {
        DWAConfig adapted = base_config_;

        // 1. 根据障碍物密度调整
        double obstacle_density = calculate_obstacle_density(state, obstacles);
        if (obstacle_density > 0.5) {
            // 密集障碍物环境：更保守
            adapted.distance_weight *= 1.5;
            adapted.velocity_weight *= 0.7;
            adapted.predict_time *= 1.2;
        }

        // 2. 根据到目标距离调整
        double goal_distance = std::hypot(goal.x - state.x,
                                         goal.y - state.y);
        if (goal_distance < 1.0) {
            // 接近目标：更精确
            adapted.heading_weight *= 1.5;
            adapted.velocity_weight *= 0.5;
        }

        // 3. 根据当前速度调整
        if (state.v > 0.5) {
            // 高速运动：增加预测时间
            adapted.predict_time = 2.5;
        } else {
            // 低速运动：减少预测时间以提高灵活性
            adapted.predict_time = 1.0;
        }

        return adapted;
    }

    double calculate_obstacle_density(const RobotState& state,
                                      const std::vector& obstacles) {
        const double detection_radius = 3.0;  // 3米检测半径
        int nearby_obstacles = 0;

        for (const auto& obs : obstacles) {
            double dist = obs.distance_to_point(state.x, state.y);
            if (dist < detection_radius) {
                nearby_obstacles++;
            }
        }

        // 归一化到[0, 1]
        return std::min(1.0, nearby_obstacles / 10.0);
    }
};

5.3 参数调优工作流

// 参数调优辅助工具
class DWATuner {
public:
    struct TuningResult {
        DWAConfig config;
        double success_rate;
        double avg_time;
        double avg_path_length;
        double avg_min_clearance;
    };

    TuningResult grid_search_tuning(
        const std::vector& scenarios) {

        std::vector results;

        // 网格搜索参数空间
        for (double hw = 0.5; hw <= 2.0; hw += 0.5) {
            for (double dw = 1.0; dw <= 3.0; dw += 0.5) {
                for (double vw = 0.1; vw <= 1.0; vw += 0.3) {
                    DWAConfig config = base_config_;
                    config.heading_weight = hw;
                    config.distance_weight = dw;
                    config.velocity_weight = vw;

                    TuningResult result = evaluate_config(config, scenarios);
                    results.push_back(result);
                }
            }
        }

        // 选择最佳配置
        return *std::max_element(results.begin(), results.end(),
            [](const TuningResult& a, const TuningResult& b) {
                return a.success_rate < b.success_rate;
            });
    }

private:
    TuningResult evaluate_config(const DWAConfig& config,
                                 const std::vector& scenarios) {
        TuningResult result;
        result.config = config;

        int successes = 0;
        double total_time = 0.0;
        double total_length = 0.0;
        double total_clearance = 0.0;

        for (const auto& scenario : scenarios) {
            auto metrics = run_scenario(config, scenario);
            if (metrics.success) {
                successes++;
                total_time += metrics.time;
                total_length += metrics.path_length;
                total_clearance += metrics.min_clearance;
            }
        }

        int n = scenarios.size();
        result.success_rate = static_cast(successes) / n;
        result.avg_time = total_time / successes;
        result.avg_path_length = total_length / successes;
        result.avg_min_clearance = total_clearance / successes;

        return result;
    }
};

6. 狭窄空间处理

6.1 狭窄通道检测

class NarrowSpaceHandler {
private:
    double narrow_threshold_ = 0.8;  // 通道宽度阈值（米）

public:
    bool is_narrow_space(const RobotState& state,
                        const std::vector& obstacles) {
        // 在机器人前方检测通道宽度
        const double check_distance = 1.5;
        const int num_rays = 9;

        std::vector distances;

        for (int i = 0; i < num_rays; ++i) {
            double angle = state.theta - M_PI/4 +
                          (M_PI/2) * i / (num_rays - 1);

            double dist = ray_cast(state.x, state.y, angle,
                                  check_distance, obstacles);
            distances.push_back(dist);
        }

        // 计算通道宽度
        double left_dist = *std::min_element(
            distances.begin(), distances.begin() + num_rays/2);
        double right_dist = *std::min_element(
            distances.begin() + num_rays/2 + 1, distances.end());

        double passage_width = left_dist + right_dist;

        return passage_width < narrow_threshold_;
    }

    // 射线检测
    double ray_cast(double x, double y, double angle,
                   double max_distance,
                   const std::vector& obstacles) {
        const double step = 0.05;
        double dist = 0.0;

        while (dist < max_distance) {
            double check_x = x + dist * std::cos(angle);
            double check_y = y + dist * std::sin(angle);

            for (const auto& obs : obstacles) {
                if (obs.contains_point(check_x, check_y)) {
                    return dist;
                }
            }

            dist += step;
        }

        return max_distance;
    }
};

6.2 狭窄空间策略调整

class NarrowSpaceDWA : public DWAPlanner {
private:
    NarrowSpaceHandler narrow_handler_;

public:
    std::pair compute_velocity_smart(
        const RobotState& state,
        const std::vector& obstacles,
        const Point& goal) {

        // 检测是否在狭窄空间
        bool is_narrow = narrow_handler_.is_narrow_space(state, obstacles);

        if (is_narrow) {
            return compute_velocity_narrow(state, obstacles, goal);
        } else {
            return compute_velocity(state, obstacles, goal);
        }
    }

private:
    std::pair compute_velocity_narrow(
        const RobotState& state,
        const std::vector& obstacles,
        const Point& goal) {

        // 狭窄空间专用配置
        DWAConfig narrow_config = config_;

        // 1. 降低速度上限
        narrow_config.max_v = 0.3;  // 限制最大线速度

        // 2. 提高避障权重
        narrow_config.distance_weight = 3.0;

        // 3. 减小预测时间（提高灵活性）
        narrow_config.predict_time = 0.8;

        // 4. 增加采样密度
        narrow_config.v_resolution = 0.02;
        narrow_config.w_resolution = 0.05;

        // 5. 优先直行，限制角速度
        narrow_config.max_w = 0.5;

        // 使用调整后的配置规划
        return compute_velocity_with_config(state, obstacles, goal,
                                           narrow_config);
    }

    // 添加朝向对齐策略
    std::pair align_with_passage(
        const RobotState& state,
        const std::vector& obstacles) {

        // 计算通道方向
        double passage_angle = estimate_passage_direction(state, obstacles);

        // 计算角度差
        double angle_diff = normalize_angle(passage_angle - state.theta);

        // 如果角度差较大，先旋转对齐
        if (std::abs(angle_diff) > M_PI / 6) {  // 30度
            return {0.05, 0.3 * std::copysign(1.0, angle_diff)};
        }

        // 已对齐，正常规划
        return compute_velocity_narrow(state, obstacles, goal_);
    }

    double estimate_passage_direction(const RobotState& state,
                                      const std::vector& obstacles) {
        // 使用PCA或其他方法估计通道方向
        // 这里简化为取前方可行方向的平均
        std::vector feasible_angles;

        for (double angle = -M_PI/2; angle <= M_PI/2; angle += 0.1) {
            double check_angle = state.theta + angle;
            double dist = narrow_handler_.ray_cast(
                state.x, state.y, check_angle, 2.0, obstacles);

            if (dist > 0.5) {  // 可行方向
                feasible_angles.push_back(check_angle);
            }
        }

        // 返回平均角度
        if (feasible_angles.empty()) {
            return state.theta;
        }

        double sum = std::accumulate(feasible_angles.begin(),
                                    feasible_angles.end(), 0.0);
        return sum / feasible_angles.size();
    }
};

6.3 死锁检测与恢复

class DeadlockRecovery {
private:
    std::deque state_history_;
    const size_t history_size_ = 20;
    const double stuck_threshold_ = 0.1;  // 位置变化阈值

public:
    bool is_stuck(const RobotState& current_state) {
        state_history_.push_back(current_state);

        if (state_history_.size() > history_size_) {
            state_history_.pop_front();
        }

        if (state_history_.size() < history_size_) {
            return false;
        }

        // 检查位置变化
        double total_displacement = 0.0;
        for (size_t i = 1; i < state_history_.size(); ++i) {
            double dx = state_history_[i].x - state_history_[i-1].x;
            double dy = state_history_[i].y - state_history_[i-1].y;
            total_displacement += std::hypot(dx, dy);
        }

        double avg_displacement = total_displacement / (history_size_ - 1);

        return avg_displacement < stuck_threshold_;
    }

    std::pair recovery_behavior(const RobotState& state) {
        // 恢复策略：原地旋转或后退

        // 策略1：小幅后退
        if (state.v > 0) {
            return {-0.2, 0.0};  // 后退
        }

        // 策略2：旋转寻找出路
        return {0.0, 0.5};  // 原地旋转
    }

    void reset() {
        state_history_.clear();
    }
};

7. ROS集成实现

完整的DWA规划器ROS实现： 

#include <ros/ros.h>
#include <geometry_msgs/Twist.h>
#include <nav_msgs/Odometry.h>
#include <sensor_msgs/LaserScan.h>
#include <visualization_msgs/MarkerArray.h>

class DWALocalPlanner {
private:
    ros::NodeHandle nh_;

    // Publishers
    ros::Publisher cmd_vel_pub_;
    ros::Publisher trajectory_viz_pub_;

    // Subscribers
    ros::Subscriber odom_sub_;
    ros::Subscriber scan_sub_;
    ros::Subscriber goal_sub_;

    // DWA核心组件
    DWAPlanner planner_;
    RobotState current_state_;
    std::vector obstacles_;
    Point goal_;

    // 死锁恢复
    DeadlockRecovery deadlock_recovery_;

public:
    DWALocalPlanner() {
        // 初始化发布器
        cmd_vel_pub_ = nh_.advertise<geometry_msgs::Twist>(
            "/cmd_vel", 1);
        trajectory_viz_pub_ = nh_.advertise<visualization_msgs::MarkerArray>(
            "/dwa_trajectories", 1);

        // 初始化订阅器
        odom_sub_ = nh_.subscribe("/odom", 1,
            &DWALocalPlanner::odomCallback, this);
        scan_sub_ = nh_.subscribe("/scan", 1,
            &DWALocalPlanner::scanCallback, this);
        goal_sub_ = nh_.subscribe("/move_base_simple/goal", 1,
            &DWALocalPlanner::goalCallback, this);

        // 加载参数
        load_parameters();
    }

    void load_parameters() {
        DWAConfig& config = planner_.get_config();

        nh_.param("predict_time", config.predict_time, 2.0);
        nh_.param("dt", config.dt, 0.1);
        nh_.param("v_resolution", config.v_resolution, 0.05);
        nh_.param("w_resolution", config.w_resolution, 0.1);
        nh_.param("heading_weight", config.heading_weight, 1.0);
        nh_.param("distance_weight", config.distance_weight, 2.0);
        nh_.param("velocity_weight", config.velocity_weight, 0.5);
    }

    void odomCallback(const nav_msgs::Odometry::ConstPtr& msg) {
        // 更新机器人状态
        current_state_.x = msg->pose.pose.position.x;
        current_state_.y = msg->pose.pose.position.y;

        // 从四元数提取航向角
        double qz = msg->pose.pose.orientation.z;
        double qw = msg->pose.pose.orientation.w;
        current_state_.theta = std::atan2(2.0 * qw * qz,
                                          1.0 - 2.0 * qz * qz);

        current_state_.v = msg->twist.twist.linear.x;
        current_state_.w = msg->twist.twist.angular.z;
    }

    void scanCallback(const sensor_msgs::LaserScan::ConstPtr& msg) {
        // 将激光数据转换为障碍物表示
        obstacles_.clear();

        for (size_t i = 0; i < msg->ranges.size(); ++i) {
            float range = msg->ranges[i];

            if (range < msg->range_min || range > msg->range_max) {
                continue;
            }

            double angle = msg->angle_min + i * msg->angle_increment;
            double global_angle = current_state_.theta + angle;

            // 转换到全局坐标
            double obs_x = current_state_.x + range * std::cos(global_angle);
            double obs_y = current_state_.y + range * std::sin(global_angle);

            obstacles_.push_back(Obstacle(obs_x, obs_y, 0.1));
        }
    }

    void goalCallback(const geometry_msgs::PoseStamped::ConstPtr& msg) {
        goal_.x = msg->pose.position.x;
        goal_.y = msg->pose.position.y;

        ROS_INFO("New goal received: (%.2f, %.2f)", goal_.x, goal_.y);
        deadlock_recovery_.reset();
    }

    void plan_and_control() {
        // 检查是否到达目标
        double dist_to_goal = std::hypot(goal_.x - current_state_.x,
                                        goal_.y - current_state_.y);

        if (dist_to_goal < 0.2) {
            // 到达目标，停止
            publish_zero_velocity();
            ROS_INFO("Goal reached!");
            return;
        }

        // 检查死锁
        if (deadlock_recovery_.is_stuck(current_state_)) {
            ROS_WARN("Robot stuck, executing recovery behavior");
            auto [v, w] = deadlock_recovery_.recovery_behavior(current_state_);
            publish_velocity(v, w);
            return;
        }

        // 执行DWA规划
        auto [v, w] = planner_.compute_velocity(
            current_state_, obstacles_, goal_);

        // 发布速度指令
        publish_velocity(v, w);

        // 可视化轨迹
        visualize_trajectories();
    }

    void publish_velocity(double v, double w) {
        geometry_msgs::Twist cmd_vel;
        cmd_vel.linear.x = v;
        cmd_vel.angular.z = w;
        cmd_vel_pub_.publish(cmd_vel);
    }

    void publish_zero_velocity() {
        publish_velocity(0.0, 0.0);
    }

    void visualize_trajectories() {
        // 可视化采样的轨迹
        visualization_msgs::MarkerArray markers;
        // ... 实现轨迹可视化
        trajectory_viz_pub_.publish(markers);
    }
};

int main(int argc, char** argv) {
    ros::init(argc, argv, "dwa_local_planner");

    DWALocalPlanner planner;

    ros::Rate rate(10);  // 10 Hz
    while (ros::ok()) {
        ros::spinOnce();
        planner.plan_and_control();
        rate.sleep();
    }

    return 0;
}

8.1 并行计算

#include <omp.h>

class ParallelDWA : public DWAPlanner {
public:
    std::pair compute_velocity_parallel(
        const RobotState& state,
        const std::vector& obstacles,
        const Point& goal) {

        auto [v_min, v_max, w_min, w_max] = calculate_dynamic_window(state);

        // 生成速度样本
        std::vector velocity_samples;
        for (double v = v_min; v <= v_max; v += config_.v_resolution) {
            for (double w = w_min; w <= w_max; w += config_.w_resolution) {
                velocity_samples.push_back({v, w});
            }
        }

        // 并行评估
        double best_score = -INFINITY;
        Velocity best_velocity;

        #pragma omp parallel
        {
            double local_best_score = -INFINITY;
            Velocity local_best_velocity;

            #pragma omp for nowait
            for (size_t i = 0; i < velocity_samples.size(); ++i) {
                const auto& vel = velocity_samples[i];

                Trajectory traj = predict_trajectory(state, vel);

                if (is_collision(traj, obstacles)) {
                    continue;
                }

                double score = evaluate_trajectory(traj, goal, vel);

                if (score > local_best_score) {
                    local_best_score = score;
                    local_best_velocity = vel;
                }
            }

            #pragma omp critical
            {
                if (local_best_score > best_score) {
                    best_score = local_best_score;
                    best_velocity = local_best_velocity;
                }
            }
        }

        return {best_velocity.v, best_velocity.w};
    }
};

// CUDA核函数：并行轨迹评估
__global__ void evaluate_trajectories_kernel(
    const double* v_samples,
    const double* w_samples,
    int num_samples,
    const RobotState state,
    const Obstacle* obstacles,
    int num_obstacles,
    const Point goal,
    const DWAConfig config,
    double* scores) {

    int idx = blockIdx.x * blockDim.x + threadIdx.x;

    if (idx >= num_samples) return;

    double v = v_samples[idx];
    double w = w_samples[idx];

    // 预测轨迹并评估
    double score = 0.0;
    RobotState current = state;
    bool collision = false;

    for (double t = 0; t <= config.predict_time; t += config.dt) {
        // 运动学模型
        if (abs(w) > 1e-6) {
            double r = v / w;
            current.x += r * (sin(current.theta + w * config.dt) -
                             sin(current.theta));
            current.y += r * (-cos(current.theta + w * config.dt) +
                             cos(current.theta));
            current.theta += w * config.dt;
        } else {
            current.x += v * cos(current.theta) * config.dt;
            current.y += v * sin(current.theta) * config.dt;
        }

        // 碰撞检测
        for (int i = 0; i < num_obstacles; ++i) {
            double dist = sqrt((current.x - obstacles[i].x) *
                              (current.x - obstacles[i].x) +
                              (current.y - obstacles[i].y) *
                              (current.y - obstacles[i].y));
            if (dist < config.robot_radius + config.safety_margin) {
                collision = true;
                break;
            }
        }

        if (collision) break;
    }

    if (!collision) {
        // 计算评分
        double goal_angle = atan2(goal.y - current.y, goal.x - current.x);
        double heading_score = M_PI - abs(goal_angle - current.theta);

        // ... 其他评分项

        score = config.heading_weight * heading_score;
    }

    scores[idx] = collision ? -INFINITY : score;
}

优势与局限

优势：

  • 计算效率高，满足实时性要求（通常< 10ms）
  • 考虑机器人动力学约束，生成可执行轨迹
  • 参数直观，易于调优
  • 局部避障能力强，适应动态环境

局限性：

  • 局部规划算法，可能陷入局部最优
  • 狭窄空间通过能力有限
  • 需要配合全局规划器使用
  • 参数设置对性能影响较大
