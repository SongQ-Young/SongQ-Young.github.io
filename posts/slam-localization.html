基于激光雷达的SLAM定位技术详解

📅 2024-11-28 ✍️ SongQ-Young 🏷️ SLAM
SLAM 定位 激光雷达

1. 引言

同步定位与建图（SLAM, Simultaneous Localization and Mapping）是移动机器人领域的核心技术之一。激光雷达（LiDAR）作为高精度的测距传感器，在SLAM系统中发挥着关键作用。本文将系统性地介绍基于激光雷达的SLAM技术，重点讲解Cartographer和LOAM两大主流算法，并提供实战开发与调试经验。

激光雷达SLAM相比于视觉SLAM具有以下优势：

  • 精度高：毫米级测距精度，不受光照影响
  • 鲁棒性强：在弱纹理、光照变化环境下表现稳定
  • 实时性好：点云处理速度快，适合实时建图
  • 直接测距：无需通过图像计算深度信息

2. SLAM系统概述

2.1 SLAM问题的数学描述

SLAM问题的本质是在未知环境中，机器人需要同时估计自身位姿和构建环境地图。数学上可以表示为： 

# 状态估计问题
# x_t: t时刻的机器人位姿 (x, y, θ)
# m: 环境地图
# z_t: t时刻的传感器观测
# u_t: t时刻的控制输入

# 后验概率
P(x_{0:t}, m | z_{1:t}, u_{1:t})

# 贝叶斯滤波
P(x_t, m | z_{1:t}, u_{1:t}) =
    η · P(z_t | x_t, m) · ∫ P(x_t | x_{t-1}, u_t) · P(x_{t-1}, m | z_{1:t-1}, u_{1:t-1}) dx_{t-1}

2.2 SLAM系统架构

一个完整的SLAM系统通常包含以下模块：

  • 前端（Frontend）：数据关联、特征提取、位姿估计
  • 后端（Backend）：图优化、位姿校正
  • 回环检测（Loop Closure）：检测并消除累积误差
  • 建图（Mapping）：构建栅格地图或点云地图
// SLAM系统基本框架
class SLAMSystem {
public:
    void processFrame(const LaserScan& scan, const Odometry& odom) {
        // 1. 前端处理：扫描匹配
        Pose2D pose_estimate = frontend_.scanMatch(scan, odom);

        // 2. 添加到位姿图
        Node* node = pose_graph_.addNode(pose_estimate, scan);

        // 3. 回环检测
        if (loop_detector_.detectLoop(node)) {
            Constraint loop_constraint = loop_detector_.computeConstraint();
            pose_graph_.addConstraint(loop_constraint);

            // 4. 后端优化
            backend_.optimizeGraph(pose_graph_);
        }

        // 5. 更新地图
        map_builder_.updateMap(pose_estimate, scan);
    }

private:
    Frontend frontend_;
    PoseGraph pose_graph_;
    LoopDetector loop_detector_;
    Backend backend_;
    MapBuilder map_builder_;
};

3. 激光雷达传感器

3.1 2D激光雷达

2D LiDAR在水平面进行扫描，常用于室内AGV和服务机器人：

  • SICK TiM系列：270°扫描角，25m测距，50Hz扫描频率
  • Hokuyo URG系列：240°扫描角，30m测距，40Hz扫描频率
  • RPLIDAR系列：360°扫描角，12m测距，10Hz扫描频率
// ROS中读取2D激光数据
#include <sensor_msgs/LaserScan.h>

void laserCallback(const sensor_msgs::LaserScan::ConstPtr& scan) {
    float angle_min = scan->angle_min;
    float angle_max = scan->angle_max;
    float angle_increment = scan->angle_increment;

    for (size_t i = 0; i < scan->ranges.size(); i++) {
        float range = scan->ranges[i];
        float angle = angle_min + i * angle_increment;

        // 转换为笛卡尔坐标
        if (range > scan->range_min && range < scan->range_max) {
            float x = range * cos(angle);
            float y = range * sin(angle);
            // 处理点云数据
            processPoint(x, y);
        }
    }
}

4. 2D SLAM算法

4.1 扫描匹配（Scan Matching）

扫描匹配是2D SLAM的核心技术，通过对齐连续扫描来估计机器人位姿。

ICP算法（Iterative Closest Point）

#include <Eigen/Dense>

class ICP {
public:
    Eigen::Matrix3f align(const PointCloud& source,
                          const PointCloud& target,
                          int max_iterations = 50,
                          float tolerance = 1e-6) {
        Eigen::Matrix3f transform = Eigen::Matrix3f::Identity();
        PointCloud transformed = source;

        for (int iter = 0; iter < max_iterations; iter++) {
            // 1. 寻找最近点对应
            std::vector<std::pair<int, int>> correspondences;
            for (size_t i = 0; i < transformed.size(); i++) {
                int nearest = findNearestPoint(transformed[i], target);
                correspondences.push_back({i, nearest});
            }

            // 2. 计算变换矩阵
            Eigen::Matrix3f delta_transform =
                computeTransform(transformed, target, correspondences);

            // 3. 应用变换
            transform = delta_transform * transform;
            transformed = applyTransform(source, transform);

            // 4. 检查收敛
            if (transformNorm(delta_transform) < tolerance) {
                break;
            }
        }

        return transform;
    }

private:
    int findNearestPoint(const Point& p, const PointCloud& cloud) {
        float min_dist = std::numeric_limits<float>::max();
        int nearest_idx = -1;

        for (size_t i = 0; i < cloud.size(); i++) {
            float dist = (p - cloud[i]).norm();
            if (dist < min_dist) {
                min_dist = dist;
                nearest_idx = i;
            }
        }

        return nearest_idx;
    }
};

相关性扫描匹配（Correlative Scan Matching）

class CorrelativeScanMatcher {
public:
    Pose2D match(const LaserScan& scan,
                 const OccupancyGrid& map,
                 const Pose2D& initial_pose) {
        float best_score = -std::numeric_limits<float>::max();
        Pose2D best_pose = initial_pose;

        // 在搜索窗口内遍历
        for (float dx = -search_window_; dx <= search_window_; dx += resolution_) {
            for (float dy = -search_window_; dy <= search_window_; dy += resolution_) {
                for (float dtheta = -angle_window_; dtheta <= angle_window_; dtheta += angle_resolution_) {
                    Pose2D candidate(
                        initial_pose.x + dx,
                        initial_pose.y + dy,
                        initial_pose.theta + dtheta
                    );

                    // 计算匹配分数
                    float score = computeScore(scan, map, candidate);

                    if (score > best_score) {
                        best_score = score;
                        best_pose = candidate;
                    }
                }
            }
        }

        return best_pose;
    }

private:
    float computeScore(const LaserScan& scan,
                      const OccupancyGrid& map,
                      const Pose2D& pose) {
        float score = 0.0;

        for (const auto& point : scan.points) {
            // 将点转换到地图坐标系
            Point2D map_point = transformPoint(point, pose);

            // 查询地图概率
            float probability = map.getProbability(map_point);
            score += log(probability);
        }

        return score;
    }

    float search_window_ = 0.5;  // 0.5米搜索窗口
    float resolution_ = 0.05;    // 5cm分辨率
    float angle_window_ = 0.3;   // ±0.3弧度
    float angle_resolution_ = 0.01;
};

4.2 栅格地图构建

class OccupancyGridMapper {
public:
    void updateMap(const LaserScan& scan, const Pose2D& pose) {
        for (const auto& endpoint : scan.points) {
            // 将激光终点转换到地图坐标系
            Point2D map_endpoint = transformPoint(endpoint, pose);

            // 射线起点（机器人位置）
            Point2D map_origin(pose.x, pose.y);

            // 射线追踪：更新空闲区域
            bresenham(map_origin, map_endpoint, [this](int x, int y) {
                updateCell(x, y, FREE_SPACE_PROB);
            });

            // 更新占用区域
            int end_x = worldToGrid(map_endpoint.x);
            int end_y = worldToGrid(map_endpoint.y);
            updateCell(end_x, end_y, OCCUPIED_PROB);
        }
    }

private:
    void updateCell(int x, int y, float observation_prob) {
        if (!isInBounds(x, y)) return;

        // 贝叶斯更新
        float prior_prob = grid_[y][x];
        float prior_odds = prior_prob / (1.0f - prior_prob);
        float observation_odds = observation_prob / (1.0f - observation_prob);

        float posterior_odds = prior_odds * observation_odds;
        float posterior_prob = posterior_odds / (1.0f + posterior_odds);

        grid_[y][x] = posterior_prob;
    }

    void bresenham(Point2D start, Point2D end,
                   std::function<void(int, int)> callback) {
        int x0 = worldToGrid(start.x);
        int y0 = worldToGrid(start.y);
        int x1 = worldToGrid(end.x);
        int y1 = worldToGrid(end.y);

        int dx = abs(x1 - x0);
        int dy = abs(y1 - y0);
        int sx = (x0 < x1) ? 1 : -1;
        int sy = (y0 < y1) ? 1 : -1;
        int err = dx - dy;

        while (x0 != x1 || y0 != y1) {
            callback(x0, y0);

            int e2 = 2 * err;
            if (e2 > -dy) {
                err -= dy;
                x0 += sx;
            }
            if (e2 < dx) {
                err += dx;
                y0 += sy;
            }
        }
    }

    std::vector<std::vector<float>> grid_;
    const float FREE_SPACE_PROB = 0.3;
    const float OCCUPIED_PROB = 0.7;
};

5. Google Cartographer算法

5.1 Cartographer架构

Cartographer是Google开源的2D/3D SLAM系统，采用图优化方法，支持多传感器融合。其核心特点包括：

  • Local SLAM：实时扫描匹配和子图构建
  • Global SLAM：回环检测和全局优化
  • Submap机制：将地图分为多个子图，提高效率
  • 多传感器融合：支持IMU、里程计等传感器

6. LOAM算法（3D SLAM）

6.1 LOAM原理

LOAM（Lidar Odometry and Mapping）是实时3D激光SLAM的经典算法，将SLAM问题分解为高频里程计估计和低频地图优化两个并行进程。

特征提取

LOAM从点云中提取边缘点（edge points）和平面点（planar points）： 

class FeatureExtractor {
public:
    void extractFeatures(const pcl::PointCloud<PointType>::Ptr& cloud) {
        edge_points_.clear();
        planar_points_.clear();

        // 按扫描线组织点云
        std::vector<std::vector<PointType>> scan_lines =
            organizeByScanLines(cloud);

        for (const auto& scan_line : scan_lines) {
            for (size_t i = 5; i < scan_line.size() - 5; i++) {
                // 计算曲率
                float curvature = computeCurvature(scan_line, i);

                // 边缘点：高曲率
                if (curvature > edge_threshold_) {
                    edge_points_.push_back(scan_line[i]);
                }
                // 平面点：低曲率
                else if (curvature < planar_threshold_) {
                    planar_points_.push_back(scan_line[i]);
                }
            }
        }

        // 降采样
        downSample(edge_points_, 0.2);
        downSample(planar_points_, 0.4);
    }

private:
    float computeCurvature(const std::vector<PointType>& points, int idx) {
        const auto& center = points[idx];
        Eigen::Vector3f sum(0, 0, 0);

        // 计算局部邻域的离散度
        for (int j = -5; j <= 5; j++) {
            if (j == 0) continue;
            const auto& neighbor = points[idx + j];
            sum.x() += neighbor.x - center.x;
            sum.y() += neighbor.y - center.y;
            sum.z() += neighbor.z - center.z;
        }

        return sum.norm();
    }

    pcl::PointCloud<PointType>::Ptr edge_points_;
    pcl::PointCloud<PointType>::Ptr planar_points_;
    float edge_threshold_ = 0.1;
    float planar_threshold_ = 0.01;
};

点到边缘和平面的距离

// 点到线的距离（边缘特征）
double pointToEdgeDistance(const Eigen::Vector3d& point,
                           const Eigen::Vector3d& line_p1,
                           const Eigen::Vector3d& line_p2) {
    Eigen::Vector3d a = point - line_p1;
    Eigen::Vector3d b = line_p2 - line_p1;
    Eigen::Vector3d cross = a.cross(b);
    return cross.norm() / b.norm();
}

// 点到平面的距离（平面特征）
double pointToPlaneDistance(const Eigen::Vector3d& point,
                            const Eigen::Vector3d& plane_p1,
                            const Eigen::Vector3d& plane_p2,
                            const Eigen::Vector3d& plane_p3) {
    Eigen::Vector3d v1 = plane_p2 - plane_p1;
    Eigen::Vector3d v2 = plane_p3 - plane_p1;
    Eigen::Vector3d normal = v1.cross(v2);
    normal.normalize();

    Eigen::Vector3d p = point - plane_p1;
    return fabs(p.dot(normal));
}

// LM优化求解位姿
class LOAMOdometry {
public:
    Eigen::Matrix4d estimateTransform() {
        // Levenberg-Marquardt优化
        Eigen::Matrix transform = Eigen::Matrix::Zero();

        for (int iter = 0; iter < 25; iter++) {
            Eigen::Matrix H = Eigen::Matrix::Zero();
            Eigen::Matrix b = Eigen::Matrix::Zero();

            // 边缘特征残差
            for (const auto& correspondence : edge_correspondences_) {
                addEdgeResidual(correspondence, H, b, transform);
            }

            // 平面特征残差
            for (const auto& correspondence : planar_correspondences_) {
                addPlanarResidual(correspondence, H, b, transform);
            }

            // 求解增量
            Eigen::Matrix delta = H.ldlt().solve(-b);
            transform += delta;

            if (delta.norm() < 1e-6) break;
        }

        return transformVector6dToMatrix4d(transform);
    }

private:
    void addEdgeResidual(const EdgeCorrespondence& corr,
                        Eigen::Matrix& H,
                        Eigen::Matrix& b,
                        const Eigen::Matrix& transform) {
        // 计算雅可比矩阵和残差
        Eigen::Vector3d transformed_point = transformPoint(corr.point, transform);
        double residual = pointToEdgeDistance(transformed_point, corr.line_p1, corr.line_p2);

        Eigen::Matrix jacobian = computeEdgeJacobian(corr, transform);

        H += jacobian.transpose() * jacobian;
        b += jacobian.transpose() * residual;
    }

    std::vector<EdgeCorrespondence> edge_correspondences_;
    std::vector<PlanarCorrespondence> planar_correspondences_;
};

6.3 LeGO-LOAM优化

LeGO-LOAM（Lightweight and Ground-Optimized LOAM）针对地面机器人进行优化： 

class LeGOLOAM {
private:
    // 1. 地面分割
    void groundSegmentation(const pcl::PointCloud<PointType>::Ptr& cloud) {
        for (int j = 0; j < N_SCAN; j++) {
            for (int i = 0; i < Horizon_SCAN; i++) {
                int lower_idx = j*Horizon_SCAN + i;
                int upper_idx = (j+1)*Horizon_SCAN + i;

                // 计算垂直角度差
                float diff_z = cloud->points[upper_idx].z - cloud->points[lower_idx].z;
                float diff_range = pointDistance(cloud->points[upper_idx],
                                                cloud->points[lower_idx]);
                float angle = atan2(diff_z, diff_range) * 180 / M_PI;

                // 地面点判断
                if (abs(angle) <= 10) {
                    ground_mat_.at<int8_t>(j, i) = 1;
                    ground_cloud_->push_back(cloud->points[lower_idx]);
                }
            }
        }
    }

    // 2. 点云分割
    void cloudSegmentation() {
        for (int j = 0; j < N_SCAN; j++) {
            for (int i = 0; i < Horizon_SCAN; i++) {
                if (ground_mat_.at<int8_t>(j, i) == 1) continue;

                // 连通性检测
                std::vector<int> queue;
                queue.push_back(j*Horizon_SCAN + i);

                while (!queue.empty()) {
                    // BFS搜索
                    int idx = queue.front();
                    queue.erase(queue.begin());

                    // 标记分割
                    label_mat_.at<int>(idx) = label_count_;

                    // 搜索邻域
                    std::vector<int> neighbors = findNeighbors(idx);
                    for (int neighbor : neighbors) {
                        if (label_mat_.at<int>(neighbor) == 0) {
                            queue.push_back(neighbor);
                        }
                    }
                }

                label_count_++;
            }
        }
    }

    cv::Mat ground_mat_;
    cv::Mat label_mat_;
    int label_count_ = 1;
    pcl::PointCloud<PointType>::Ptr ground_cloud_;
};

7. 传感器融合与回环检测

7.1 IMU融合

class IMUIntegration {
public:
    void integrateIMU(const sensor_msgs::Imu& imu_msg) {
        double dt = (imu_msg.header.stamp - last_imu_time_).toSec();

        // 加速度积分得到速度
        velocity_ += acceleration_ * dt;

        // 速度积分得到位置
        position_ += velocity_ * dt + 0.5 * acceleration_ * dt * dt;

        // 角速度积分得到姿态（四元数）
        Eigen::Quaterniond delta_q =
            Eigen::AngleAxisd(angular_velocity_.x() * dt, Eigen::Vector3d::UnitX()) *
            Eigen::AngleAxisd(angular_velocity_.y() * dt, Eigen::Vector3d::UnitY()) *
            Eigen::AngleAxisd(angular_velocity_.z() * dt, Eigen::Vector3d::UnitZ());

        orientation_ = orientation_ * delta_q;
        orientation_.normalize();

        // 更新加速度和角速度
        acceleration_ = Eigen::Vector3d(
            imu_msg.linear_acceleration.x,
            imu_msg.linear_acceleration.y,
            imu_msg.linear_acceleration.z
        );

        angular_velocity_ = Eigen::Vector3d(
            imu_msg.angular_velocity.x,
            imu_msg.angular_velocity.y,
            imu_msg.angular_velocity.z
        );

        last_imu_time_ = imu_msg.header.stamp;
    }

    Eigen::Isometry3d getPose() {
        Eigen::Isometry3d pose = Eigen::Isometry3d::Identity();
        pose.translate(position_);
        pose.rotate(orientation_);
        return pose;
    }

private:
    Eigen::Vector3d position_;
    Eigen::Vector3d velocity_;
    Eigen::Vector3d acceleration_;
    Eigen::Vector3d angular_velocity_;
    Eigen::Quaterniond orientation_;
    ros::Time last_imu_time_;
};

7.2 回环检测

#include <opencv2/opencv.hpp>
#include <DBoW3/DBoW3.h>

class LoopClosureDetector {
public:
    bool detectLoop(const pcl::PointCloud<PointType>::Ptr& current_cloud,
                    int current_frame_id,
                    int& matched_frame_id) {
        // 1. 提取描述子（可以使用Scan Context或M2DP）
        Eigen::VectorXf descriptor = extractDescriptor(current_cloud);
        descriptors_.push_back(descriptor);

        // 2. 搜索相似帧（排除近期帧）
        float min_distance = std::numeric_limits<float>::max();
        matched_frame_id = -1;

        for (int i = 0; i < current_frame_id - 100; i++) {
            float distance = (descriptor - descriptors_[i]).norm();

            if (distance < min_distance && distance < loop_threshold_) {
                min_distance = distance;
                matched_frame_id = i;
            }
        }

        if (matched_frame_id < 0) return false;

        // 3. ICP精确配准验证
        pcl::PointCloud<PointType>::Ptr matched_cloud =
            getCloudByFrameId(matched_frame_id);

        pcl::IterativeClosestPoint<PointType, PointType> icp;
        icp.setInputSource(current_cloud);
        icp.setInputTarget(matched_cloud);
        icp.setMaxCorrespondenceDistance(1.0);
        icp.setMaximumIterations(100);
        icp.setTransformationEpsilon(1e-6);
        icp.setEuclideanFitnessEpsilon(1e-6);

        pcl::PointCloud<PointType>::Ptr aligned(new pcl::PointCloud<PointType>);
        icp.align(*aligned);

        // 4. 验证匹配质量
        if (icp.hasConverged() && icp.getFitnessScore() < fitness_threshold_) {
            loop_transform_ = icp.getFinalTransformation().cast<double>();
            return true;
        }

        return false;
    }

    Eigen::Matrix4d getLoopTransform() const {
        return loop_transform_;
    }

private:
    Eigen::VectorXf extractDescriptor(const pcl::PointCloud<PointType>::Ptr& cloud) {
        // Scan Context描述子
        const int NUM_RINGS = 20;
        const int NUM_SECTORS = 60;

        Eigen::MatrixXf scan_context = Eigen::MatrixXf::Zero(NUM_RINGS, NUM_SECTORS);

        for (const auto& point : cloud->points) {
            float range = sqrt(point.x*point.x + point.y*point.y);
            float angle = atan2(point.y, point.x);

            int ring_idx = std::min(int(range / max_range_ * NUM_RINGS), NUM_RINGS - 1);
            int sector_idx = int((angle + M_PI) / (2*M_PI) * NUM_SECTORS);

            scan_context(ring_idx, sector_idx) =
                std::max(scan_context(ring_idx, sector_idx), point.z);
        }

        // 压缩为一维向量
        Eigen::VectorXf descriptor(NUM_RINGS * NUM_SECTORS);
        for (int i = 0; i < NUM_RINGS; i++) {
            for (int j = 0; j < NUM_SECTORS; j++) {
                descriptor(i * NUM_SECTORS + j) = scan_context(i, j);
            }
        }

        return descriptor;
    }

    std::vector<Eigen::VectorXf> descriptors_;
    Eigen::Matrix4d loop_transform_;
    float loop_threshold_ = 0.15;
    float fitness_threshold_ = 0.3;
    float max_range_ = 80.0;
};

9.2 自动驾驶场景应用

// 高速场景下的3D SLAM
class HighSpeedSLAM {
public:
    void processFrame(const sensor_msgs::PointCloud2::ConstPtr& msg) {
        // 1. 点云预处理
        pcl::PointCloud<PointXYZI>::Ptr raw_cloud(new pcl::PointCloud<PointXYZI>);
        pcl::fromROSMsg(*msg, *raw_cloud);

        // 运动补偿（去畸变）
        motionCompensation(raw_cloud);

        // 地面去除
        pcl::PointCloud<PointXYZI>::Ptr non_ground(new pcl::PointCloud<PointXYZI>);
        removeGround(raw_cloud, non_ground);

        // 2. 特征提取
        pcl::PointCloud<PointXYZI>::Ptr edge_features(new pcl::PointCloud<PointXYZI>);
        pcl::PointCloud<PointXYZI>::Ptr plane_features(new pcl::PointCloud<PointXYZI>);
        extractFeatures(non_ground, edge_features, plane_features);

        // 3. 位姿估计
        Eigen::Matrix4d pose = estimatePose(edge_features, plane_features);

        // 4. 地图更新（异步）
        map_update_thread_ = std::thread([this, non_ground, pose]() {
            updateMap(non_ground, pose);
        });

        // 5. 发布位姿
        publishPose(pose);
    }

private:
    void motionCompensation(pcl::PointCloud<PointXYZI>::Ptr& cloud) {
        // 利用IMU数据进行运动补偿
        double scan_period = 0.1;  // 100ms扫描周期

        for (auto& point : cloud->points) {
            // 假设点云中存储了相对时间戳
            double ratio = point.intensity / 100.0;  // 归一化时间

            // 插值计算该点对应的位姿
            Eigen::Isometry3d interp_pose = interpolatePose(ratio);

            // 变换到扫描结束时刻
            Eigen::Vector3d p(point.x, point.y, point.z);
            p = interp_pose.inverse() * p;

            point.x = p.x();
            point.y = p.y();
            point.z = p.z();
        }
    }

    std::thread map_update_thread_;
};
